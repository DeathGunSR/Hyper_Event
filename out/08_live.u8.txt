   CREATE PROCEDURE [dbo].[sp_Res_ShowFacture_Fac_List]

   @F_Code  bigint,

   @Fac_Type VarChar(1),

   @FType_KF Varchar(1)

    AS

   Select F.*, FL.*, A.A_Code, A.A_Code_C, A.A_Name,A.WeightUnit,A.Place From Res_Facture F

   Left Join Res_Fac_List FL On (F.F_Code = FL.F_Code )

   Left join Article A On (A.A_Code = FL.A_Code)

   Where F.F_Code = @F_Code and F.Fac_Type= @Fac_Type and F.FType_KF= @FType_KF
